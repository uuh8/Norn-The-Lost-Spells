// Surface Nets固定Slot配合紧凑Index：每个采样Cube至多一个顶点，每条Core Edge至多一张Quad。
// Halo重复计算同一World点，不在Chunk边界使用单独不同的采样或颜色赢家。
struct SurfaceVertex { float3 position;float valid;float3 normal;uint weights0;uint3 weights;uint pad; };
RWStructuredBuffer<float> _Field;
RWStructuredBuffer<SurfaceVertex> _Vertices;
RWStructuredBuffer<float3> _PositionsA,_PositionsB;
RWStructuredBuffer<uint> _Indices,_Counters,_Arguments;
StructuredBuffer<float4> _OwnerBounds;
int _NodeSide,_CubeSide,_IndexCapacity,_SmoothPass;
// OwnerKeys.w是本帧工作Mask。缓存页继续Draw；所有阶段都早退，保留同一套
// Field/Vertex/Index/Indirect Args，不能只跳过Field却把旧页的Counter清零。
bool UpdatePage(uint page){return _OwnerKeys[page].w!=0;}
void PrepareAlpha(uint page){SampleAlpha=(_OwnerKeys[page].w&1073741824)!=0?1:_Alpha;}
int3 Coord(uint i,int side) { int3 c;c.x=i%side;i/=side;c.y=i%side;c.z=i/side;return c; }
uint LocalIndex(int3 c,int side) { return c.x+side*(c.y+side*c.z); }
// 半采样间距偏移避免一层满格的上下边界都恰好采到零值。每页使用同一World格点相位。
float3 WorldCell(uint page,int3 node) { return (float3)_OwnerKeys[page].xyz*_K+((float3)node-3+.5)/_Resolution; }
float Node(uint page,int3 c) { return _Field[page*_NodeSide*_NodeSide*_NodeSide+LocalIndex(c,_NodeSide)]; }
static const int3 Corners[8]={int3(0,0,0),int3(1,0,0),int3(0,1,0),int3(1,1,0),int3(0,0,1),int3(1,0,1),int3(0,1,1),int3(1,1,1)};
static const int2 Edges[12]={int2(0,1),int2(2,3),int2(4,5),int2(6,7),int2(0,2),int2(1,3),int2(4,6),int2(5,7),int2(0,4),int2(1,5),int2(2,6),int2(3,7)};
void SampleFieldWork(uint3 tid)
{
    uint slots=_NodeSide*_NodeSide*_NodeSide;if(tid.x>=slots*_OwnerCount)return;
    uint page=tid.x/slots;if(!UpdatePage(page))return;PrepareAlpha(page);int3 c=Coord(tid.x%slots,_NodeSide);float3 p=WorldCell(page,c);
    float3 low=_OwnerBounds[page*2].xyz,high=_OwnerBounds[page*2+1].xyz;
    _Field[tid.x]=(any(p<low) || any(p>high))?-.65:Field(p);
}
// 同一算法的线程组对照：不改节点位置/数量/相位，实测较小Group是否改善寄存器调度。
[numthreads(64,1,1)]void SampleField(uint3 tid:SV_DispatchThreadID){SampleFieldWork(tid);}
[numthreads(32,1,1)]void SampleField32(uint3 tid:SV_DispatchThreadID){SampleFieldWork(tid);}
[numthreads(64,1,1)]void BuildVertices(uint3 tid:SV_DispatchThreadID)
{
    uint slots=_CubeSide*_CubeSide*_CubeSide;if(tid.x>=slots*_OwnerCount)return;
    uint page=tid.x/slots;if(!UpdatePage(page))return;int3 c=Coord(tid.x%slots,_CubeSide);float f[8];uint mask=0;
    [unroll]for(int i=0;i<8;i++){f[i]=Node(page,c+Corners[i]);if(f[i]>0)mask|=1u<<i;}
    SurfaceVertex vertex=(SurfaceVertex)0;float3 position=0;int count=0;
    if(mask!=0 && mask!=255)
    {
        [unroll]for(int i=0;i<12;i++){int2 e=Edges[i];if((f[e.x]>0)==(f[e.y]>0))continue;
            float t=saturate(f[e.x]/(f[e.x]-f[e.y]));position+=lerp((float3)Corners[e.x],(float3)Corners[e.y],t);count++;}
        position=WorldCell(page,c)+position/(max(1,count)*_Resolution);vertex.valid=1;
        vertex.normal=normalize(-float3(f[1]+f[3]+f[5]+f[7]-f[0]-f[2]-f[4]-f[6],
            f[2]+f[3]+f[6]+f[7]-f[0]-f[1]-f[4]-f[5],f[4]+f[5]+f[6]+f[7]-f[0]-f[1]-f[2]-f[3])+float3(0,1e-8,0));
    }
    vertex.position=position;_Vertices[tid.x]=vertex;_PositionsA[tid.x]=position;
}
[numthreads(64,1,1)]void SmoothVertices(uint3 tid:SV_DispatchThreadID)
{
    uint slots=_CubeSide*_CubeSide*_CubeSide;if(tid.x>=slots*_OwnerCount)return;
    uint page=tid.x/slots;if(!UpdatePage(page))return;int3 c=Coord(tid.x%slots,_CubeSide);float3 at=_SmoothPass==0?_PositionsA[tid.x]:_PositionsB[tid.x],sum=0;int count=0;
    if(_Vertices[tid.x].valid>0 && all(c>0) && all(c<_CubeSide-1))
    {
        [unroll]for(int axis=0;axis<3;axis++)[unroll]for(int sign=-1;sign<=1;sign+=2)
        {int3 n=c;n[axis]+=sign;uint id=page*slots+LocalIndex(n,_CubeSide);if(_Vertices[id].valid<=0)continue;
            sum+=_SmoothPass==0?_PositionsA[id]:_PositionsB[id];count++;}
        // 两次Taubin迭代一收一放：圆润棱角，同时抑制单纯Laplacian造成液量剪影缩小。
        if(count>0)at+=(_SmoothPass==0?.5:-.53)*(sum/count-at);
        float3 low=WorldCell(page,c);at=clamp(at,low,low+1.0/_Resolution);
    }
    if(_SmoothPass==0)_PositionsB[tid.x]=at;else _PositionsA[tid.x]=at;
}
uint HalfPair(float a,float b){return f32tof16(a)|(f32tof16(b)<<16);}
float LocalField(uint page,int3 c,float3 q,out float3 gradient)
{
    float value=0;gradient=0;
    [unroll]for(int i=0;i<8;i++)
    {
        int3 bit=Corners[i];float3 w=lerp(1-q,q,(float3)bit);float f=Node(page,c+bit);
        value+=f*w.x*w.y*w.z;
        gradient+=f*float3((bit.x*2-1)*w.y*w.z,(bit.y*2-1)*w.x*w.z,(bit.z*2-1)*w.x*w.y)*_Resolution;
    }
    return value;
}
void FinishVerticesWork(uint3 tid)
{
    uint slots=_CubeSide*_CubeSide*_CubeSide;if(tid.x>=slots*_OwnerCount)return;
    uint page=tid.x/slots;if(!UpdatePage(page))return;PrepareAlpha(page);SurfaceVertex v=_Vertices[tid.x];if(v.valid<=0)return;
    int3 c=Coord(tid.x%slots,_CubeSide);float3 low=WorldCell(page,c),p=_PositionsA[tid.x],gradient;
    // Taubin之后用已有8个节点的Trilinear场投影。每步限制在原Cube，保持页边界一致，
    // 避免平滑把等值面推离实际数量；这里没有额外175邻格Kernel查询。
    [unroll]for(int projection=0;projection<3;projection++)
    {float value=LocalField(page,c,saturate((p-low)*_Resolution),gradient);p=clamp(p-gradient*value/max(1e-6,dot(gradient,gradient)),low,low+1.0/_Resolution);}
    LocalField(page,c,saturate((p-low)*_Resolution),gradient);
    v.normal=dot(gradient,gradient)>1e-10?-normalize(gradient):v.normal;
    float swellWeight;CompressGroundProxy(p,v.normal,swellWeight);v.pad=asuint(swellWeight);
    float4 a,b;MaterialWeights(p,a,b);
    v.position=_OriginSize.xyz+p*_OriginSize.w;v.weights0=HalfPair(a.x,a.y);v.weights=uint3(HalfPair(a.z,a.w),HalfPair(b.x,b.y),HalfPair(b.z,b.w));_Vertices[tid.x]=v;
    // 第二次平滑已结束，复用PositionsB保存最终World Space位置，不新增GPU Buffer。
    _PositionsB[tid.x]=v.position;
}
[numthreads(64,1,1)]void FinishVertices(uint3 tid:SV_DispatchThreadID){FinishVerticesWork(tid);}
[numthreads(32,1,1)]void FinishVertices32(uint3 tid:SV_DispatchThreadID){FinishVerticesWork(tid);}
float3 QuadNormal(uint page,int3 c,int axis,uint vertexId)
{
    int3 next=c;next[axis]++;float a=Node(page,c),b=Node(page,next);if((a>0)==(b>0))return 0;
    int u=(axis+1)%3,w=(axis+2)%3;int3 c1=c,c2=c,c3=c;c1[u]--;c2[u]--;c2[w]--;c3[w]--;
    uint slots=_CubeSide*_CubeSide*_CubeSide,baseSlot=page*slots;
    uint i0=baseSlot+LocalIndex(c,_CubeSide),i1=baseSlot+LocalIndex(c1,_CubeSide),i2=baseSlot+LocalIndex(c2,_CubeSide),i3=baseSlot+LocalIndex(c3,_CubeSide);
    if(_Vertices[i0].valid<=0 || _Vertices[i1].valid<=0 || _Vertices[i2].valid<=0 || _Vertices[i3].valid<=0)return 0;
    float3 p0=_PositionsB[i0],p1=_PositionsB[i1],p2=_PositionsB[i2],p3=_PositionsB[i3];
    // 一个Quad的两张Triangle可能不共面；只累计真正引用本Vertex的Triangle。
    // 把另一张面也平均进去会在极薄侧壁上抵消顶面Normal。
    float3 areaNormal=0;
    if(vertexId==i0 || vertexId==i1 || vertexId==i2)areaNormal+=cross(p1-p0,p2-p0);
    if(vertexId==i0 || vertexId==i2 || vertexId==i3)areaNormal+=cross(p2-p0,p3-p0);
    return a>0?areaNormal:-areaNormal;
}
[numthreads(64,1,1)]void FinishNormals(uint3 tid:SV_DispatchThreadID)
{
    uint slots=_CubeSide*_CubeSide*_CubeSide;if(tid.x>=slots*_OwnerCount)return;
    uint page=tid.x/slots;if(!UpdatePage(page))return;SurfaceVertex vertex=_Vertices[tid.x];if(vertex.valid<=0)return;
    int3 c=Coord(tid.x%slots,_CubeSide);if(any(c<1) || any(c>=_CubeSide-1))return;
    // 薄的倾斜等值面在同一个Cube中同时靠近顶/底；Trilinear梯度会指向底面。
    // 在形变之后按真实绕序累加相邻12张Quad的面积法线；包括Halo面，跨页结果一致。
    // 独立Dispatch确保所有位置已写完，不增加Kernel场查询或Float Atomic。
    float3 normal=0;
    [unroll]for(int axis=0;axis<3;axis++)[unroll]for(int u=0;u<2;u++)[unroll]for(int w=0;w<2;w++)
    {int3 edge=c;edge[(axis+1)%3]+=u;edge[(axis+2)%3]+=w;normal+=QuadNormal(page,edge,axis,tid.x);}
    if(dot(normal,normal)>1e-16)vertex.normal=normalize(normal);
    _Vertices[tid.x]=vertex;
}
void Face(uint page,int3 c,int axis)
{
    int3 next=c;next[axis]++;float a=Node(page,c),b=Node(page,next);if((a>0)==(b>0))return;
    int u=(axis+1)%3,v=(axis+2)%3;int3 c1=c,c2=c,c3=c;c1[u]--;c2[u]--;c2[v]--;c3[v]--;
    uint slots=_CubeSide*_CubeSide*_CubeSide,baseSlot=page*slots;
    uint ids[4]={baseSlot+LocalIndex(c,_CubeSide),baseSlot+LocalIndex(c1,_CubeSide),baseSlot+LocalIndex(c2,_CubeSide),baseSlot+LocalIndex(c3,_CubeSide)};
    [unroll]for(int i=0;i<4;i++)if(_Vertices[ids[i]].valid<=0)return;
    uint offset;InterlockedAdd(_Counters[page],6,offset);if(offset+6>(uint)_IndexCapacity)return;
    uint start=page*_IndexCapacity+offset;
    // 当前Quad循环法线是+axis；内部正值→外部负值沿+axis时保持绕序。
    if(a>0){_Indices[start]=ids[0];_Indices[start+1]=ids[1];_Indices[start+2]=ids[2];_Indices[start+3]=ids[0];_Indices[start+4]=ids[2];_Indices[start+5]=ids[3];}
    else{_Indices[start]=ids[0];_Indices[start+1]=ids[2];_Indices[start+2]=ids[1];_Indices[start+3]=ids[0];_Indices[start+4]=ids[3];_Indices[start+5]=ids[2];}
}
[numthreads(64,1,1)]void BuildFaces(uint3 tid:SV_DispatchThreadID)
{
    int n=_K*_Resolution;uint slots=n*n*n;if(tid.x>=slots*_OwnerCount)return;
    uint page=tid.x/slots;if(!UpdatePage(page))return;int3 c=Coord(tid.x%slots,n)+3;Face(page,c,0);Face(page,c,1);Face(page,c,2);
}
[numthreads(64,1,1)]void ClearCounters(uint3 tid:SV_DispatchThreadID){if(tid.x<(uint)_OwnerCount && UpdatePage(tid.x))_Counters[tid.x]=0;}
[numthreads(64,1,1)]void FinishArguments(uint3 tid:SV_DispatchThreadID)
{
    if(tid.x>=(uint)_OwnerCount || !UpdatePage(tid.x))return;uint at=tid.x*5;_Arguments[at]=min(_Counters[tid.x],(uint)_IndexCapacity);
    _Arguments[at+1]=1;_Arguments[at+2]=tid.x*_IndexCapacity;_Arguments[at+3]=0;_Arguments[at+4]=0;
}
